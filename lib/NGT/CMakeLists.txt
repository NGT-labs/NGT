if(UNIX OR WIN32)
	option(NGT_SHARED_MEMORY_ALLOCATOR "enable shared memory" OFF)
	configure_file(${CMAKE_CURRENT_SOURCE_DIR}/defines.h.in ${CMAKE_CURRENT_BINARY_DIR}/defines.h)
	include_directories("${CMAKE_CURRENT_BINARY_DIR}" "${PROJECT_SOURCE_DIR}/lib" "${PROJECT_BINARY_DIR}/lib/")

	if(${NGT_QBG_DISABLED})
		file(GLOB NGT_SOURCES *.cpp)
	else()
		file(GLOB NGT_SOURCES *.cpp NGTQ/*.cpp)
	endif()
	file(GLOB HEADER_FILES *.h *.hpp)
	file(GLOB NGTQ_HEADER_FILES NGTQ/*.h NGTQ/*.hpp)

	add_library(ngtstatic STATIC ${NGT_SOURCES})
	set_target_properties(ngtstatic PROPERTIES OUTPUT_NAME ngt)
	set_target_properties(ngtstatic PROPERTIES COMPILE_FLAGS "-fPIC")
	target_link_libraries(ngtstatic)
	if(CMAKE_CXX_COMPILER_ID STREQUAL "AppleClang")
		target_link_libraries(ngtstatic OpenMP::OpenMP_CXX)
	endif()

	add_library(ngt SHARED ${NGT_SOURCES})
	set_target_properties(ngt PROPERTIES VERSION ${ngt_VERSION})
	set_target_properties(ngt PROPERTIES SOVERSION ${ngt_SOVERSION})
	add_dependencies(ngt ngtstatic)
	if((NOT DEFINED NGT_QBG_DISABLED) OR (NOT ${NGT_QBG_DISABLED}))
		if((DEFINED NGT_X86SIMDSORT) OR (DEFINED NGTQG_X86SIMDSORT))
			target_link_libraries(ngt lapack blas x86simdsortcpp)
		else()
			target_link_libraries(ngt lapack blas)
		endif()
	endif()
	if(${APPLE})
		if(CMAKE_CXX_COMPILER_ID STREQUAL "AppleClang")
			target_link_libraries(ngt OpenMP::OpenMP_CXX)
		else()
			target_link_libraries(ngt gomp)
		endif()
	else(${APPLE})
		target_link_libraries(ngt gomp rt)
		if(NGT_FOREST)
			target_link_libraries(ngt stdc++fs)
			target_link_libraries(ngtstatic stdc++fs)
		endif()
	endif(${APPLE})
	# --- Generate version_defs.h at configure time natively ---
	string(TIMESTAMP NGT_BUILD_DATE "%Y/%m/%d %H:%M:%S")
	find_package(Git QUIET)
	if(GIT_FOUND)
		execute_process(COMMAND ${GIT_EXECUTABLE} log -1 --format=%H WORKING_DIRECTORY ${PROJECT_SOURCE_DIR} OUTPUT_VARIABLE NGT_GIT_HASH OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
		execute_process(COMMAND ${GIT_EXECUTABLE} log -1 --format=%cd WORKING_DIRECTORY ${PROJECT_SOURCE_DIR} OUTPUT_VARIABLE NGT_GIT_DATE OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
		execute_process(COMMAND ${GIT_EXECUTABLE} describe --abbrev=0 WORKING_DIRECTORY ${PROJECT_SOURCE_DIR} OUTPUT_VARIABLE NGT_GIT_TAG OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
	endif()
	if(EXISTS "${PROJECT_SOURCE_DIR}/VERSION")
		file(READ "${PROJECT_SOURCE_DIR}/VERSION" NGT_VERSION)
		string(STRIP "${NGT_VERSION}" NGT_VERSION)
	endif()
	
	set(VERSION_DEFS_CONTENT "//\n// Do *NOT* edit this file.\n//\n")
	if(NGT_BUILD_DATE)
		set(VERSION_DEFS_CONTENT "${VERSION_DEFS_CONTENT}#define\tNGT_BUILD_DATE\t\t\"${NGT_BUILD_DATE}\"\n")
	endif()
	if(NGT_GIT_HASH)
		set(VERSION_DEFS_CONTENT "${VERSION_DEFS_CONTENT}#define\tNGT_GIT_HASH\t\t\"${NGT_GIT_HASH}\"\n")
	endif()
	if(NGT_GIT_DATE)
		set(VERSION_DEFS_CONTENT "${VERSION_DEFS_CONTENT}#define\tNGT_GIT_DATE\t\t\"${NGT_GIT_DATE}\"\n")
	endif()
	if(NGT_GIT_TAG)
		set(VERSION_DEFS_CONTENT "${VERSION_DEFS_CONTENT}#define\tNGT_GIT_TAG\t\t\"${NGT_GIT_TAG}\"\n")
	endif()
	if(NGT_VERSION)
		set(VERSION_DEFS_CONTENT "${VERSION_DEFS_CONTENT}#define\tNGT_VERSION\t\t\"${NGT_VERSION}\"\n")
	endif()

	file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/version_defs.h.in" "${VERSION_DEFS_CONTENT}")
	configure_file("${CMAKE_CURRENT_BINARY_DIR}/version_defs.h.in" "${CMAKE_CURRENT_BINARY_DIR}/version_defs.h" COPYONLY)
	install(TARGETS
		ngt
		ngtstatic
		RUNTIME DESTINATION bin
		LIBRARY DESTINATION lib
		ARCHIVE DESTINATION lib)
	install(FILES ${HEADER_FILES} ${CMAKE_CURRENT_BINARY_DIR}/defines.h  ${CMAKE_CURRENT_BINARY_DIR}/version_defs.h DESTINATION include/NGT)
	install(FILES ${NGTQ_HEADER_FILES} DESTINATION include/NGT/NGTQ)
endif()
